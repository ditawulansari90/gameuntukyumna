    let currentModuleIndex = 0;

    function colorizeSyllables(text) {
      const words = text.split(' ');
      return words.map(word => {
        // Simple syllable splitter for KVKV / KVKVK patterns (2-3 chars per chunk)
        if (word.length <= 3) {
          return `<span class="text-pink-600">${word}</span>`;
        } else if (word.length === 4) { // e.g. ba-ju
          return `<span class="text-pink-600">${word.slice(0, 2)}</span><span class="text-indigo-600">${word.slice(2)}</span>`;
        } else if (word.length === 5) { // e.g. ma-can
          return `<span class="text-pink-600">${word.slice(0, 2)}</span><span class="text-indigo-600">${word.slice(2)}</span>`;
        } else if (word.length === 6) { // e.g. se-ka-li
          return `<span class="text-pink-600">${word.slice(0, 2)}</span><span class="text-indigo-600">${word.slice(2, 4)}</span><span class="text-teal-600">${word.slice(4)}</span>`;
        } else {
          return `<span class="text-slate-800">${word}</span>`;
        }
      }).join(' ');
    }

    function renderModule(index) {
      currentModuleIndex = index;
      const mod = modulesData[index];

      // Update Header & Dropdown
      document.getElementById('module-select').value = index;
      document.getElementById('current-mod-num').innerText = `Modul ${index + 1}`;
      document.getElementById('story-title').innerText = mod.title;

      // Render Progress Dots
      const dotsContainer = document.getElementById('progress-dots');
      dotsContainer.innerHTML = modulesData.map((m, i) => `
        <div class="w-3 h-3 rounded-full transition-all duration-300 ${i === index ? 'bg-pink-500 w-6' : 'bg-pink-200'}"></div>
      `).join('');

      // Render Illustration
      document.getElementById('illustration-container').innerHTML = mod.svg;

      // Render Story Text with Syllables
      const storyLinesContainer = document.getElementById('story-lines');
      storyLinesContainer.innerHTML = mod.lines.map(line => `
        <p class="leading-relaxed font-semibold">${colorizeSyllables(line)}</p>
      `).join('');

      // Render Questions Form
      const qContainer = document.getElementById('questions-container');
      qContainer.innerHTML = mod.questions.map((q, qIdx) => {
        let inputHTML = q.inputs.map(inp => `
          <input type="text" id="${inp.id}" 
            autocomplete="off"
            placeholder="..."
            class="w-28 sm:w-32 text-center px-3 py-1.5 bg-purple-50/70 border-2 border-purple-200 focus:border-purple-500 focus:bg-white text-purple-900 font-bold rounded-xl outline-none transition duration-200 text-lg">
        `).join(' ');

        return `
          <div class="flex flex-wrap items-center gap-2 text-slate-700 font-bold text-lg md:text-xl bg-slate-50 p-3.5 rounded-2xl border border-slate-100">
            <span>${q.text}</span>
            ${inputHTML}
          </div>
        `;
      }).join('');

      // Reset Quiz Result View
      const resDiv = document.getElementById('quiz-result');
      resDiv.classList.add('hidden');
      resDiv.className = "hidden w-full sm:w-auto p-4 rounded-2xl text-center flex items-center justify-center space-x-3 transition-all duration-300";

      // Disable/Enable Navigation Buttons
      document.getElementById('btn-prev').disabled = (index === 0);
      document.getElementById('btn-prev').style.opacity = (index === 0) ? "0.5" : "1";

      document.getElementById('btn-next').disabled = (index === modulesData.length - 1);
      document.getElementById('btn-next').style.opacity = (index === modulesData.length - 1) ? "0.5" : "1";
    }

    function playSpeech() {
      if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel(); // Stop active speech
        const mod = modulesData[currentModuleIndex];
        const utterance = new SpeechSynthesisUtterance(mod.speechText);
        utterance.lang = 'id-ID';
        utterance.rate = 0.8; // Friendly slow speed for kids
        utterance.pitch = 1.1;
        window.speechSynthesis.speak(utterance);
      } else {
        alert("Perangkat Anda tidak mendukung fitur suara bawaan (Speech Synthesis).");
      }
    }

    function checkAnswers() {
      const mod = modulesData[currentModuleIndex];
      let totalInputs = 0;
      let correctInputs = 0;

      mod.questions.forEach(q => {
        q.inputs.forEach(inp => {
          totalInputs++;
          const el = document.getElementById(inp.id);
          const val = el.value.toLowerCase().trim();

          if (val === inp.ans.toLowerCase()) {
            correctInputs++;
            el.classList.remove('border-purple-200', 'border-red-400', 'bg-red-50');
            el.classList.add('border-emerald-500', 'bg-emerald-50', 'text-emerald-800');
          } else {
            el.classList.remove('border-purple-200', 'border-emerald-500', 'bg-emerald-50');
            el.classList.add('border-red-400', 'bg-red-50', 'text-red-800');
          }
        });
      });

      // Display Result Badge
      const resDiv = document.getElementById('quiz-result');
      resDiv.classList.remove('hidden');

      if (correctInputs === totalInputs) {
        resDiv.className = "w-full sm:w-auto p-4 rounded-2xl text-center flex items-center justify-center space-x-2 bg-emerald-500 text-white font-heading font-bold text-lg shadow-lg bounce-subtle";
        resDiv.innerHTML = `
          <span>⭐⭐⭐ Hebat Sekali! Semua Jawaban Benar! 🎉</span>
        `;
      } else {
        resDiv.className = "w-full sm:w-auto p-4 rounded-2xl text-center flex items-center justify-center space-x-2 bg-amber-500 text-white font-heading font-bold text-lg shadow-md";
        resDiv.innerHTML = `
          <span>Nilai: ${correctInputs} / ${totalInputs} • Coba periksa lagi yang berwarna merah ya!</span>
        `;
      }
    }

    function switchModule(index) {
      renderModule(parseInt(index));
    }

    function navigateModule(direction) {
      const newIndex = currentModuleIndex + direction;
      if (newIndex >= 0 && newIndex < modulesData.length) {
        renderModule(newIndex);
        window.scrollTo({ top: 0, behavior: 'smooth' });
      }
    }

    // Initialize Application
    window.onload = function() {
      renderModule(0);
    };
